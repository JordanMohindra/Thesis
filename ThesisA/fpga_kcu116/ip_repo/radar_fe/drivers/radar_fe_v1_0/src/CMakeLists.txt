cmake_minimum_required(VERSION 3.15)
project(radar_fe)
find_package(common)
collector_create (PROJECT_LIB_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}")
collect (PROJECT_LIB_SOURCES xradar_fe.c)
collect (PROJECT_LIB_SOURCES xradar_fe_sinit.c)
collect (PROJECT_LIB_SOURCES xradar_fe_linux.c)
collect (PROJECT_LIB_SOURCES xradar_fe_g.c)
collector_list (_sources PROJECT_LIB_SOURCES)
collector_create (PROJECT_LIB_HEADERS "${CMAKE_CURRENT_SOURCE_DIR}")
collect (PROJECT_LIB_HEADERS xradar_fe.h)
collect (PROJECT_LIB_HEADERS xradar_fe_hw.h)
collector_list (_headers PROJECT_LIB_HEADERS)
file(COPY ${_headers} DESTINATION ${CMAKE_BINARY_DIR}/include)
include_directories(${CMAKE_BINARY_DIR}/include)
add_library(radar_fe STATIC ${_sources})
set_target_properties(radar_fe PROPERTIES LINKER_LANGUAGE C)
